# Commands to run on your production server

# 1. Copy the script to the server (run this from your LOCAL machine)
scp create_api_key.py regadmin@mail.aelira.ai:/opt/aelira-backend/

# 2. SSH into the server
ssh regadmin@mail.aelira.ai

# 3. Run the script inside the API container
cd /opt/aelira-backend
docker exec -it aelira-backend-api python create_api_key.py

# The script will output your API key - COPY IT IMMEDIATELY!
